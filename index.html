Secure PII Collector

  • AWS
  • API Gateway
  • Lambda
  • CloudFront
  • DynamoDB

Personal information like names and Social Security numbers is a prime target. A collection form needs to reject forged or tampered requests, stop malformed data before it's stored, and hold up against request floods.

  • A serverless web app that collects PII and protects it from the edge to the database.
  • HMAC authentication, so only signed requests are accepted.
  • Server-side request validation before anything is written.
  • API throttling and security controls for reliability under load.
  • CloudFront serves the app over HTTPS at the edge.
  • API Gateway routes requests and applies throttling.
  • Lambda verifies the HMAC signature and validates the payload.
  • DynamoDB stores records only after both checks pass.

// Sign: HMAC-SHA256 over the exact request body (Web Crypto API)
async function hmacHex(key, message) {
  const enc = new TextEncoder();
  const k = await crypto.subtle.importKey(
    "raw", enc.encode(key), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const sig = await crypto.subtle.sign("HMAC", k, enc.encode(message));
  return [...new Uint8Array(sig)].map(b => b.toString(16).padStart(2, "0")).join("");
}

// Verify: compare every byte so response time doesn't leak where it differs
function timingSafeEqual(a, b) {
  if (a.length !== b.length) return false;
  let diff = 0;
  for (let i = 0; i < a.length; i++) diff |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return diff === 0;
}

// Rate limit: token bucket refilled continuously
function take(bucket, now) {
  bucket.tokens = Math.min(bucket.capacity,
    bucket.tokens + (now - bucket.last) / 1000 * bucket.ratePerSec);
  bucket.last = now;
  if (bucket.tokens < 1) return false;   // 429 Too Many Requests
  bucket.tokens -= 1;
  return true;
}

const features = (x, y) => poly ? [1, x, y, x*x, y*y, x*y] : [1, x, y];

// One epoch of batch gradient descent on binary cross-entropy
function step(points, w, lr) {
  const grad = new Array(w.length).fill(0);
  let loss = 0;
  for (const { x, y, label } of points) {
    const f = features(x, y);
    const p = sigmoid(f.reduce((z, fi, i) => z + w[i] * fi, 0));
    const err = p - label;                     // dLoss/dz
    for (let i = 0; i < w.length; i++) grad[i] += err * f[i];
    loss -= label * Math.log(p + 1e-9) + (1 - label) * Math.log(1 - p + 1e-9);
  }
  for (let i = 0; i < w.length; i++) w[i] -= lr * grad[i] / points.length;
  return loss / points.length;
}
